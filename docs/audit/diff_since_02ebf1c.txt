+"""TPL-01: Transient Conduction and Thermal Mass Verifier.
+
+Implements M5-Screen and M5-Eval per FLBDS v9 §5:
+- Screen: Stored sensible energy E = rho * cp * d * delta_T >= required;
+  Fourier number Fo = alpha * period / d^2 in [0.25, 5.0]; thickness and mass caps.
+- Eval: 24h cyclic solve with Robin boundary condition: swing attenuation (damping factor)
+  and phase lag; diurnal thermal swing within allowed interior limits.
+"""
+
+from typing import Dict, Tuple
+import numpy as np
+
+from flbds.schema.templates import TPL01Params
+from flbds.schema.verification import PhysicsCheck
+
+
+def compute_thermal_diffusivity(k: float, rho: float, cp: float) -> float:
+    """Computes thermal diffusivity alpha = k / (rho * cp) in m^2/s."""
+    if rho <= 0 or cp <= 0:
+        return 0.0
+    return k / (rho * cp)
+
+
+def compute_diurnal_damping_and_lag(
+    thickness_m: float,
+    alpha_m2_s: float,
+    period_s: float = 86400.0,
+) -> Tuple[float, float]:
+    """Computes semi-infinite / slab diurnal thermal damping factor and time lag.
+
+    Reference: Incropera & DeWitt; Carslaw & Jaeger (Conduction of Heat in Solids).
+    decay_constant k_w = sqrt(pi / (alpha * period))
+    damping_factor = exp(-k_w * thickness)
+    phase_lag_hours = (k_w * thickness) * (period_hours / (2 * pi))
+    """
+    if alpha_m2_s <= 0 or period_s <= 0 or thickness_m <= 0:
+        return 0.0, 0.0
+
+    k_w = np.sqrt(np.pi / (alpha_m2_s * period_s))
+    damping = float(np.exp(-k_w * thickness_m))
+    lag_s = float(k_w * thickness_m * period_s / (2.0 * np.pi))
+    lag_hours = lag_s / 3600.0
+    return damping, lag_hours
+
+
+def screen_tpl01(
+    params: TPL01Params,
+    delta_T_design_K: float = 10.0,
+    mass_cap_kg_m2: float = 500.0,
+    thickness_cap_m: float = 0.50,
+) -> PhysicsCheck:
+    """M5-Screen: Stored sensible energy, Fo in [0.25, 5.0], mass and thickness caps."""
+    d = params.thickness_m
+    if d > thickness_cap_m:
+        return PhysicsCheck(
+            template_name="TPL01_M5_Screen",
+            passed=False,
+            reason=f"Thickness {d*1e3:.1f} mm exceeds cap {thickness_cap_m*1e3:.1f} mm",
+            metrics={"thickness_m": d},
+        )
+
+    mass_per_area = params.density_kg_m3 * d
+    if mass_per_area > mass_cap_kg_m2:
+        return PhysicsCheck(
+            template_name="TPL01_M5_Screen",
+            passed=False,
+            reason=f"Mass per area {mass_per_area:.1f} kg/m2 exceeds cap {mass_cap_kg_m2:.1f} kg/m2",
+            metrics={"mass_per_area": mass_per_area},
+        )
+
+    # Sensible stored energy per area
+    stored_energy = mass_per_area * params.specific_heat_J_kgK * delta_T_design_K
+    if stored_energy < params.required_energy_J_m2:
+        return PhysicsCheck(
+            template_name="TPL01_M5_Screen",
+            passed=False,
+            reason=f"Sensible capacity {stored_energy/1e6:.3f} MJ/m2 below required {params.required_energy_J_m2/1e6:.3f} MJ/m2",
+            metrics={"stored_energy_J_m2": stored_energy, "required_energy_J_m2": params.required_energy_J_m2},
+        )
+
+    # Fourier number Fo = alpha * t / d^2
+    alpha = compute_thermal_diffusivity(params.thermal_conductivity_W_mK, params.density_kg_m3, params.specific_heat_J_kgK)
+    period_s = params.period_hours * 3600.0
+    Fo = (alpha * period_s) / (d ** 2) if d > 0 else 0.0
+
+    if Fo < 0.25 or Fo > 5.0:
+        return PhysicsCheck(
+            template_name="TPL01_M5_Screen",
+            passed=False,
+            reason=f"Fourier number Fo={Fo:.3f} outside valid thermal mass regime [0.25, 5.0]",
+            metrics={"Fo": Fo, "alpha_m2_s": alpha},
+        )
+
+    return PhysicsCheck(
+        template_name="TPL01_M5_Screen",
+        passed=True,
+        reason="Screen thermal mass capacity and Fourier regime satisfied",
+        metrics={"stored_energy_MJ_m2": stored_energy / 1e6, "Fo": Fo, "mass_per_area": mass_per_area},
+    )
+
+

+    )

+"""TPL-03: Evaporation and Condensation Verifier.

+import numpy as np
+
+from flbds.schema.templates import TPL03Params

+def saturation_vapor_pressure_Pa(T_degC: float) -> float:
+    """Tetens / Antoine formula for saturation vapor pressure of water in Pa."""
+    return float(610.78 * np.exp((17.27 * T_degC) / (T_degC + 237.3)))
+
+
+def humidity_ratio(vapor_pressure_Pa: float, total_pressure_Pa: float = 101325.0) -> float:
+    """Computes humidity ratio omega (kg water / kg dry air)."""
+    p_v = min(vapor_pressure_Pa, total_pressure_Pa * 0.95)
+    return float(0.622 * p_v / (total_pressure_Pa - p_v))
+
+
+def dew_point_temperature_degC(vapor_pressure_Pa: float) -> float:
+    """Inverts Tetens formula to find dew point temperature in degC."""
+    p_v = max(10.0, vapor_pressure_Pa)
+    val = np.log(p_v / 610.78)
+    return float((237.3 * val) / (17.27 - val))
+

+    Re_L = (rho_air * params.air_velocity_m_s * params.characteristic_length_m) / 1.8e-5
+    Pr = 0.71
+    Nu_L = 0.664 * (Re_L ** 0.5) * (Pr ** (1.0 / 3.0)) if Re_L < 5e5 else 0.037 * (Re_L ** 0.8) * (Pr ** (1.0 / 3.0))
+    h_c = Nu_L * 0.026 / params.characteristic_length_m

+    h_m = h_c / (rho_air * cp_air * (Le ** (2.0 / 3.0)))

+        )

+        passed=True,
+        reason="Coupled Chilton-Colburn mass transfer and evaporative cooling verified",
+        metrics={"evap_cooling_flux_W_m2": evap_cooling_flux, "latent_cooling_flux_W_m2": evap_cooling_flux, "h_m_m_s": h_m, "Re_L": Re_L},
+    )

+"""TPL-05: Capillary Wicking Verifier.
+
+Implements M5-Screen and M5-Eval per FLBDS v9 §5:
+- Screen: Jurin capillary rise height >= 1.5 * lift_height; Bond number Bo < 0.1.
+- Eval: Washburn / Darcy mass supply rate at lift height covers required evaporation/replenishment flow;
+  viscous pressure drop within capillary pressure limit; pore clearance above fouling limit.
+"""
+
+from typing import Dict, Tuple, Optional
+import numpy as np
+
+from flbds.schema.templates import TPL05Params
+from flbds.schema.verification import PhysicsCheck
+
+G_ACCEL = 9.80665  # Gravitational acceleration m/s^2
+
+
+def compute_jurin_height_m(gamma: float, theta_rad: float, rho: float, r_pore: float) -> float:
+    """Computes equilibrium Jurin capillary rise height h = (2 * gamma * cos(theta)) / (rho * g * r_pore)."""
+    if rho <= 0 or r_pore <= 0:
+        return 0.0
+    return float((2.0 * gamma * np.cos(theta_rad)) / (rho * G_ACCEL * r_pore))
+
+
+def compute_bond_number(rho: float, r_pore: float, gamma: float) -> float:
+    """Computes Bond number Bo = (rho * g * r_pore^2) / gamma."""
+    if gamma <= 0:
+        return float("inf")
+    return float((rho * G_ACCEL * (r_pore ** 2)) / gamma)
+
+
+def screen_tpl05(
+    params: TPL05Params,
+) -> PhysicsCheck:
+    """M5-Screen: Jurin height >= 1.5 * lift_height and Bond number < 0.1."""
+    theta_rad = np.radians(params.contact_angle_deg)
+    h_jurin = compute_jurin_height_m(params.surface_tension_N_m, theta_rad, params.fluid_density_kg_m3, params.pore_radius_m)
+    bo = compute_bond_number(params.fluid_density_kg_m3, params.pore_radius_m, params.surface_tension_N_m)
+
+    if h_jurin < 1.5 * params.lift_height_m:
+        return PhysicsCheck(
+            template_name="TPL05_M5_Screen",
+            passed=False,
+            reason=f"Jurin height {h_jurin:.3f} m below 1.5x required lift ({1.5*params.lift_height_m:.3f} m)",
+            metrics={"h_jurin_m": h_jurin, "required_lift_m": params.lift_height_m, "Bo": bo},
+        )
+
+    if bo >= 0.1:
+        return PhysicsCheck(
+            template_name="TPL05_M5_Screen",
+            passed=False,
+            reason=f"Bond number Bo={bo:.4f} >= 0.1 (gravity dominates over capillary force)",
+            metrics={"Bo": bo, "h_jurin_m": h_jurin},
+        )
+
+    return PhysicsCheck(
+        template_name="TPL05_M5_Screen",
+        passed=True,
+        reason=f"Screen capillary lift verified (h_jurin={h_jurin:.3f} m, Bo={bo:.4e})",
+        metrics={"h_jurin_m": h_jurin, "Bo": bo},
+    )
+
+
+def eval_tpl05(
+    params: TPL05Params,
+    min_pore_diameter_fouling_m: float = 5e-7,  # 500 nm particulate fouling ceiling
+    min_pore_diameter_m: Optional[float] = None,
+) -> PhysicsCheck:
+    """M5-Eval: Coupled Darcy-Washburn fluid replenishment supply rate and fouling clearance."""
+    if min_pore_diameter_m is not None:
+        min_pore_diameter_fouling_m = min_pore_diameter_m
+    # 1. Screen check
+    screen_res = screen_tpl05(params)
+    if not screen_res.passed:
+        return PhysicsCheck(
+            template_name="TPL05_M5_Eval",
+            passed=False,
+            reason=f"Screen check failed: {screen_res.reason}",
+            metrics=screen_res.metrics,
+        )
+
+    # 2. Pore fouling clearance check
+    pore_diam = 2.0 * params.pore_radius_m
+    if pore_diam < min_pore_diameter_fouling_m:
+        return PhysicsCheck(
+            template_name="TPL05_M5_Eval",
+            passed=False,
+            reason=f"Pore diameter {pore_diam*1e9:.1f} nm below particulate fouling limit {min_pore_diameter_fouling_m*1e9:.1f} nm",
+            metrics={"pore_diam_nm": pore_diam * 1e9, "fouling_limit_nm": min_pore_diameter_fouling_m * 1e9},
+        )
+
+    # 3. Darcy-Washburn mass flow supply rate against gravity
+    # Net driving pressure head: delta_P_cap - delta_P_grav
+    theta_rad = np.radians(params.contact_angle_deg)
+    delta_p_cap = (2.0 * params.surface_tension_N_m * np.cos(theta_rad)) / params.pore_radius_m

+            template_name="TPL05_M5_Eval",
+            passed=False,
+            reason=(
+                f"Darcy replenishment flux {mass_flux_supply*1e3:.3f} g/(m2.s) below "
+                f"required {params.required_mass_flux_kg_m2_s*1e3:.3f} g/(m2.s) (viscous drag limitation)"
+            ),
+            metrics={
+                "mass_flux_supply": mass_flux_supply,
+                "required_flux": params.required_mass_flux_kg_m2_s,
+                "delta_p_cap_Pa": delta_p_cap,
+            },
+        )
+
+    return PhysicsCheck(
+        template_name="TPL05_M5_Eval",
+        passed=True,
+        reason=(
+            f"Coupled capillary wicking verified: supply rate {mass_flux_supply*1e3:.2f} g/(m2.s) covers "
+            f"demand at {params.lift_height_m*1e3:.1f} mm lift"
+        ),
+        metrics={
+            "mass_flux_supply_kg_m2_s": mass_flux_supply,
+            "delta_p_cap_Pa": delta_p_cap,
+            "net_pressure_Pa": net_driving_pressure,
+        },
+    )

+"""TPL-06: Darcy-Forchheimer Porous Media Flow Verifier.
+
+Implements M5-Screen and M5-Eval per FLBDS v9 §5:
+- Screen: Darcy linear viscous pressure drop; pore Reynolds number Re_pore <= 10.
+- Eval: Ergun equation (coupled viscous Blake-Kozeny plus inertial Forchheimer terms);
+  pumping coefficient of performance (COP_pumping >= min_cop).
+"""
+
+from typing import Dict, Tuple
+from flbds.schema.templates import TPL06Params
+from flbds.schema.verification import PhysicsCheck
+
+
+def compute_ergun_pressure_drop(
+    L: float,
+    eps: float,
+    d_p: float,
+    rho: float,
+    mu: float,
+    u_0: float,
+) -> Tuple[float, float, float, float]:
+    """Computes viscous, inertial, and total pressure drop via the classic Ergun equation.
+
+    delta_P / L = 150 * (mu * (1 - eps)^2) / (eps^3 * d_p^2) * u_0
+                + 1.75 * (rho * (1 - eps)) / (eps^3 * d_p) * u_0^2
+
+    Returns:
+        (delta_P_total, delta_P_viscous, delta_P_inertial, Re_pore)
+    """
+    if eps <= 0 or eps >= 1 or d_p <= 0 or L <= 0:
+        return float("inf"), float("inf"), float("inf"), float("inf")
+
+    term_visc = 150.0 * (mu * ((1.0 - eps) ** 2)) / ((eps ** 3) * (d_p ** 2)) * u_0
+    term_inert = 1.75 * (rho * (1.0 - eps)) / ((eps ** 3) * d_p) * (u_0 ** 2)
+
+    dp_visc = float(term_visc * L)
+    dp_inert = float(term_inert * L)
+    dp_total = dp_visc + dp_inert
+
+    Re_pore = float((rho * u_0 * d_p) / (mu * (1.0 - eps))) if (mu * (1.0 - eps)) > 0 else 0.0
+
+    return dp_total, dp_visc, dp_inert, Re_pore
+
+
+def screen_tpl06(
+    params: TPL06Params,
+) -> PhysicsCheck:
+    """M5-Screen: Darcy linear drop and pore Reynolds number <= 10."""
+    dp_total, dp_visc, dp_inert, re_pore = compute_ergun_pressure_drop(
+        L=params.bed_length_m,
+        eps=params.bed_porosity,
+        d_p=params.mean_pore_diameter_m,
+        rho=params.fluid_density_kg_m3,
+        mu=params.fluid_viscosity_Pa_s,
+        u_0=params.superficial_velocity_m_s,
+    )
+
+    if re_pore > 10.0:
+        return PhysicsCheck(
+            template_name="TPL06_M5_Screen",
+            passed=False,
+            reason=f"Pore Reynolds number Re_p={re_pore:.2f} > 10.0 (inertial Forchheimer effects violate Darcy regime)",
+            metrics={"Re_pore": re_pore, "dp_darcy_Pa": dp_visc},
+        )
+
+    if dp_visc > params.pressure_drop_cap_Pa:
+        return PhysicsCheck(
+            template_name="TPL06_M5_Screen",
+            passed=False,
+            reason=f"Darcy pressure drop {dp_visc:.1f} Pa exceeds cap {params.pressure_drop_cap_Pa:.1f} Pa",
+            metrics={"dp_visc_Pa": dp_visc, "cap_Pa": params.pressure_drop_cap_Pa},
+        )
+
+    return PhysicsCheck(
+        template_name="TPL06_M5_Screen",
+        passed=True,
+        reason=f"Screen Darcy flow satisfied (Re_p={re_pore:.2f}, dp={dp_visc:.1f} Pa)",
+        metrics={"dp_visc_Pa": dp_visc, "Re_pore": re_pore},
+    )
+
+
+def eval_tpl06(
+    params: TPL06Params,
+    thermal_duty_W: float = 1000.0,
+    bed_cross_section_m2: float = 0.04,  # 0.2m x 0.2m
+) -> PhysicsCheck:
+    """M5-Eval: Coupled Ergun pressure drop and thermal-to-pumping COP."""
+    dp_total, dp_visc, dp_inert, re_pore = compute_ergun_pressure_drop(
+        L=params.bed_length_m,
+        eps=params.bed_porosity,
+        d_p=params.mean_pore_diameter_m,
+        rho=params.fluid_density_kg_m3,
+        mu=params.fluid_viscosity_Pa_s,
+        u_0=params.superficial_velocity_m_s,
+    )

+"""TPL-07: Extended Surface Fin Convection Verifier.
+
+Implements M5-Screen and M5-Eval per FLBDS v9 §5:
+- Screen: Straight rectangular fin efficiency eta_f = tanh(m * L_c) / (m * L_c);
+  Biot number Bi = h * (t_f / 2) / k < 0.1 (1D conduction assumption); envelope height cap.
+- Eval: Boundary-layer choking check (fin spacing S >= 2 * delta_th) or Elenbaas channel correlation;
+  coupled degraded thermal duty meets required heat rejection.
+"""
+
+from typing import Dict, Tuple
+import numpy as np
+
+from flbds.schema.templates import TPL07Params
+from flbds.schema.verification import PhysicsCheck
+
+
+def compute_constructal_spacing_m(fin_length_m: float, air_velocity_m_s: float = 2.0) -> float:
+    """Computes Bejan constructal optimal/minimum fin spacing S_opt ~ 2 * delta_th."""
+    nu_air = 1.5e-5
+    Pr = 0.71
+    delta_th = 5.0 * np.sqrt((nu_air * fin_length_m) / max(0.1, air_velocity_m_s)) / (Pr ** (1.0 / 3.0))
+    return float(2.0 * delta_th)
+
+
+def compute_fin_efficiency(h: float, k: float, t_f: float, L_f: float) -> Tuple[float, float, float]:
+    """Computes fin parameter m, corrected length L_c, and efficiency eta_f for rectangular profile.
+
+    m = sqrt(2 * h / (k * t_f))
+    L_c = L_f + t_f / 2
+    eta_f = tanh(m * L_c) / (m * L_c)
+    """
+    if k <= 0 or t_f <= 0 or L_f <= 0 or h <= 0:
+        return 0.0, 0.0, 0.0
+
+    m = np.sqrt(2.0 * h / (k * t_f))
+    L_c = L_f + (t_f / 2.0)
+    mLc = m * L_c
+    eta_f = float(np.tanh(mLc) / mLc) if mLc > 0 else 1.0
+    Bi = float((h * (t_f / 2.0)) / k)
+
+    return float(m), float(eta_f), Bi
+
+
+def screen_tpl07(
+    params: TPL07Params,
+    envelope_height_cap_m: float = 0.10,
+) -> PhysicsCheck:
+    """M5-Screen: Fin efficiency, Biot number < 0.1, envelope height cap."""
+    if params.fin_length_m > envelope_height_cap_m:
+        return PhysicsCheck(
+            template_name="TPL07_M5_Screen",
+            passed=False,
+            reason=f"Fin length {params.fin_length_m*1e3:.1f} mm exceeds envelope cap {envelope_height_cap_m*1e3:.1f} mm",
+            metrics={"fin_length_m": params.fin_length_m, "cap_m": envelope_height_cap_m},
+        )
+
+    m, eta_f, Bi = compute_fin_efficiency(
+        h=params.h_conv_W_m2K,
+        k=params.fin_conductivity_W_mK,
+        t_f=params.fin_thickness_m,
+        L_f=params.fin_length_m,
+    )
+
+    if Bi >= 0.1:
+        return PhysicsCheck(
+            template_name="TPL07_M5_Screen",
+            passed=False,
+            reason=f"Biot number Bi={Bi:.4f} >= 0.1 (violates 1D lumped-fin conduction assumption)",
+            metrics={"Bi": Bi, "eta_f": eta_f},
+        )
+
+    return PhysicsCheck(
+        template_name="TPL07_M5_Screen",
+        passed=True,
+        reason=f"Screen fin efficiency verified (eta_f={eta_f:.3f}, Bi={Bi:.4f})",
+        metrics={"eta_f": eta_f, "Bi": Bi, "m_inv_m": m},
+    )
+

+        )

+        h=params.h_conv_W_m2K,
+        k=params.fin_conductivity_W_mK,
+        t_f=params.fin_thickness_m,
+        L_f=params.fin_length_m,
+    )

+    q_total = num_fins * q_per_fin
+

+        return PhysicsCheck(
+            template_name="TPL07_M5_Eval",
+            passed=False,
+            reason=f"Total fin heat dissipation {q_total:.1f} W below required {required_duty_W:.1f} W",
+            metrics={"q_total_W": q_total, "required_W": required_duty_W, "q_per_fin": q_per_fin},
+        )
+
+    return PhysicsCheck(
+        template_name="TPL07_M5_Eval",
+        passed=True,
+        reason=f"Coupled fin convection verified: q={q_total:.1f} W >= required, spacing clear of boundary-layer choking",
+        metrics={"q_total_W": q_total, "eta_f": eta_f, "spacing_clearance_mm": (params.fin_spacing_m - 2 * delta_th) * 1e3},
+    )
